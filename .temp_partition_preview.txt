                    {/* Partition Preview */}
                    {partitions.length > 0 && (
                        <div className="space-y-2">
                            <div className="flex items-center justify-between">
                                <label className="text-sm font-medium">
                                    {t('xml_flash.partitions_preview', 'Partitions to Flash')}
                                </label>
                                <div className="flex gap-1">
                                    <Button
                                        type="button"
                                        variant="ghost"
                                        size="sm"
                                        className="h-6 text-xs"
                                        onClick={() => setSelectedPartitions(new Set(partitions.map((_, idx) => idx)))}
                                    >
                                        Select All
                                    </But>
                                    <Button
                                        type="button"
                                        variant="ghost"
                                        size="sm"
                                        className="h-6 text-xs"
                                        onClick={() => setSelectedPartitions(new Set())}
                                    >
                                        Deselect All
                                    </Button>
                                </div>
                            </div>
                            <div className="max-h-[200px] overflow-y-auto border rounded-lg">
                                <div className="divide-y">
                                    {partitions.map((p, idx) => {
                                        const sizeInMB = (parseInt(p.num_partition_sectors) * 4096 / 1024 / 1024);
                                        const sizeStr = sizeInMB >= 1024 
                                            ? `${(sizeInMB / 1024).toFixed(2)} GB`
                                            : `${sizeInMB.toFixed(2)} MB`;
                                        
                                        const lun = p.physical_partition_number || '?';
                                        const sector = p.start_sector || '?';
                                        const isSelected = selectedPartitions.has(idx);
                                        
                                        return (
                                            <label
                                                key={idx}
                                                className="px-3 py-2 flex items-start gap-2 cursor-pointer hover:bg-muted/50"
                                            >
                                                <input
                                                    type="checkbox"
                                                    checked={isSelected}
                                                    onChange={(e) => {
                                                        const newSet = new Set(selectedPartitions);
                                                        if (e.target.checked) {
                                                            newSet.add(idx);
                                                        } else {
                                                            newSet.delete(idx);
                                                        }
                                                        setSelectedPartitions(newSet);
                                                    }}
                                                    className="mt-0.5 w-4 h-4 rounded border-gray-300"
                                                />
                                                <div className="flex-1 text-sm">
                                                    <div className="flex items-center justify-between mb-1">
                                                        <div className="flex items-center gap-2">
                                                            <span className="font-mono text-xs font-semibold">{p.label}</span>
                                                           <span className="text-[10px] text-muted-foreground">
                                                                (LUN{lun} @ {sector})
                                                            </span>
                                                        </div>
                                                        <span className="text-xs text-muted-foreground">
                                                            {sizeStr}
                                                        </span>
                                                    </div>
                                                    <div className="text-xs text-muted-foreground">{p.filename}</div>
                                                </div>
                                            </label>
                                        );
                                    })}
                                </div>
                            </div>
                        </div>
                    )}
